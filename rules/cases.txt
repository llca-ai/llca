# Redaction cases: every rule in redaction.rules is proven here.
# `cargo test` runs all of them and reports the line of any case that fails.
#
#   in:   text as the agent sent it (a command, or any string argument)
#   out:  the exact text expected in the journal
#   hide: a fragment that must not appear in the journal (when the exact
#         output does not matter)
#   keep: text that must be stored unchanged (no false positive)
#   json: an argument object as JSON; follow it with out: (JSON, compared
#         as values) or hide:
#   kind: the label the dashboard shows for a masked secret ("GitHub
#         token", "URL parameter "sig"", "Value of "API_TOKEN""…)
#
# Escapes in in:/out:/hide:/keep: lines: \n (new line), \t (tab), \\ (backslash).
# json: lines (and the out: after them) use JSON's own escapes.
# A blank line or a comment ends nothing; each in:/json: starts a new case.
# Use made-up secrets only.

# ---- KEY=value, --flag value, "key": "value" ------------------------------
in:   export OPENAI_API_KEY=sk-proj-abcdefghijklmnop1234
out:  export OPENAI_API_KEY=[REDACTED]
kind: OpenAI or Anthropic API key
in:   gh auth login --token ghtoken123
hide: ghtoken123
in:   mysql --password S3cr3t
hide: S3cr3t
in:   {"credential":{"provider":"cloudflare_artifact","branch":"main","token":"art_v2_x_FAKEfake0123456789ab","auth_mode":"http_extra_header"}}
out:  {"credential":{"provider":"cloudflare_artifact","branch":"main","token":"[REDACTED]","auth_mode":"http_extra_header"}}
kind: Value of "token"
in:   echo {\"password\": \"hunter2\", \"user\": \"bob\"}
out:  echo {\"password\": \"[REDACTED]\", \"user\": \"bob\"}
in:   {'api_key': 'abc123'}
out:  {'api_key': '[REDACTED]'}
keep: {"tokens_used": 12}
keep: git commit -m 'fix secret handling'
keep: grep -r token src/
keep: grep "token" src/
